// Google Apps Script (Code.gs) function doGet(e) { var lock = LockService.getScriptLock(); lock.waitLock(10000); try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var fcstSheet = ss.getSheetByName("FCST") || ss.insertSheet("FCST"); var histSheet = ss.getSheetByName("History") || ss.insertSheet("History"); var fcstData = fcstSheet.getDataRange().getValues(); var fcstObj = {}; for (var i = 1; i < fcstData.length; i++) { if (fcstData[i][0]) { fcstObj[fcstData[i][0]] = { h: Number(fcstData[i][1]), r: Number(fcstData[i][2]) }; } } var histData = histSheet.getDataRange().getValues(); var histObj = {}; for (var i = 1; i < histData.length; i++) { if (histData[i][0]) { histObj[Number(histData[i][0])] = { h: Number(histData[i][1]), r: Number(histData[i][2]) }; } } var result = { fcst: fcstObj, history: histObj }; return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function doPost(e) { var lock = LockService.getScriptLock(); lock.waitLock(10000); try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var payload = JSON.parse(e.postData.contents); var action = payload.action; if (action === "upsert_fcst") { var sheet = ss.getSheetByName("FCST") || ss.insertSheet("FCST"); var data = sheet.getDataRange().getValues(); var rowIndex = -1; if (data.length <= 1 && (!data[0] || !data[0][0])) { sheet.getRange(1, 1, 1, 3).setValues([["담당자명", "취급액", "매출액"]]); data = sheet.getDataRange().getValues(); } for (var i = 1; i < data.length; i++) { if (data[i][0] === payload.name) { rowIndex = i + 1; break; } } if (rowIndex > -1) { sheet.getRange(rowIndex, 2).setValue(payload.handling); sheet.getRange(rowIndex, 3).setValue(payload.revenue); } else { sheet.appendRow([payload.name, payload.handling, payload.revenue]); } } else if (action === "update_history") { var sheet = ss.getSheetByName("History") || ss.insertSheet("History"); var data = sheet.getDataRange().getValues(); var rowIndex = -1; if (data.length <= 1 && (!data[0] || !data[0][0])) { sheet.getRange(1, 1, 1, 3).setValues([["월", "취급액", "매출액"]]); data = sheet.getDataRange().getValues(); } for (var i = 1; i < data.length; i++) { if (Number(data[i][0]) === Number(payload.month)) { rowIndex = i + 1; break; } } if (rowIndex > -1) { sheet.getRange(rowIndex, 2).setValue(payload.handling); sheet.getRange(rowIndex, 3).setValue(payload.revenue); } else { sheet.appendRow([payload.month, payload.handling, payload.revenue]); } } return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.toString()})).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
